set pagination off
set print pretty off

break draw_line
commands
silent
printf "[gdb] hit draw_line\n"
# print addresses and initial values
p &state
p &state.b
p &state.b.p
p state.b.p

set $i = 0
while $i < 400
  next
  printf "[gdb] step %d\n", $i
  p state.b.p
  if (state.b.p.y < -1000000 || state.b.p.y > 1000000)
    printf "[gdb] CORRUPTION detected at step %d\n", $i
    p state.b.p
    bt
    quit
  end
  set $i = $i + 1
end
continue
end
