set pagination off
set print pretty off

# Break at draw_line and inspect local 'state' and its b.p member
break draw_line
commands
silent
printf "[gdb] hit draw_line\n"
# print addresses and initial values
p &state
p &state.b
p &state.b.p
p state.b.p
# step a number of source-level steps and print the current b.p each time
set $i = 0
while $i < 40
  next
  printf "[gdb] step %d\n", $i
  p state.b.p
  set $i = $i + 1
end
quit
end

# Run the program with the sample map
run maps/00_simple.fdf
